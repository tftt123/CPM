# P3 报价流程开发指南（第一个核心业务功能）

> 技术栈：.NET 8 + EF Core + SQL Server
> 目标：实现"创建商机 → 技术评审 → 定价审批 → 发出报价"的完整流程

---

## 一、业务规则梳理

### 流程规则

1. 销售创建商机，录入客户、预计金额、报价截止日期
2. 销售在商机下创建报价单，提交给技术评审
3. 技术负责人收到待办，评估可行性和成本，填写意见
4. 技术评审通过后，流转给销售主管定价审批
5. 主管审批通过后，报价单状态变为"已发出"
6. 如果技术评审或主管审批驳回，退回上一步
7. 客户接受报价 → 赢单，进入样品阶段
8. 客户拒绝报价 → 输单，关闭商机

### 时效规则（预警用，先实现功能，预警放后面）

- 技术评审须在24小时内完成
- 定价审批须在8小时内完成
- 报价截止前3天提醒销售
- 报价截止前1天紧急提醒

`Services/QuotationService.cs`：

```csharp
using CpmServer.Common;
using CpmServer.Data;
using CpmServer.DTOs.Quotation;
using CpmServer.Models;
using Microsoft.EntityFrameworkCore;

namespace CpmServer.Services;

public class QuotationService : IQuotationService
{
    private readonly CpmDbContext _db;
    private readonly ILogger<QuotationService> _logger;

    public QuotationService(CpmDbContext db, ILogger<QuotationService> logger)
    {
        _db = db;
        _logger = logger;
    }

    public async Task<long> CreateOpportunityAsync(QuoOpportunity opp)
    {
        opp.OpportunityNo = $"OPP{DateTime.Now:yyyyMMddHHmmss}";
        opp.Stage = "NEW";
        opp.Status = 0;
        opp.CreatedAt = DateTime.Now;
        opp.UpdatedAt = DateTime.Now;

        _db.Opportunities.Add(opp);
        await _db.SaveChangesAsync();

        _logger.LogInformation("创建商机: {No}, 客户={CustomerId}", opp.OpportunityNo, opp.CustomerId);
        return opp.Id;
    }

    public async Task<long> CreateQuotationAsync(QuotationCreateDto dto, long userId)
    {
        var total = dto.Items.Sum(i => i.UnitPrice * i.Qty);

        var quote = new QuoQuotation
        {
            QuotationNo = $"Q{DateTime.Now:yyyyMMddHHmmss}",
            OpportunityId = dto.OpportunityId,
            CustomerId = dto.CustomerId,
            TotalAmount = total,
            Status = 0, // 草稿
            CreatedBy = userId,
            CreatedAt = DateTime.Now,
            UpdatedAt = DateTime.Now
        };

        _db.Quotations.Add(quote);
        await _db.SaveChangesAsync();

        // 保存明细
        foreach (var item in dto.Items)
        {
            _db.QuotationItems.Add(new QuoQuotationItem
            {
                QuotationId = quote.Id,
                ProductId = item.ProductId,
                Qty = item.Qty,
                UnitPrice = item.UnitPrice,
                LineAmount = item.UnitPrice * item.Qty
            });
        }
        await _db.SaveChangesAsync();

        _logger.LogInformation("创建报价单: {No}, 金额={Total}", quote.QuotationNo, total);
        return quote.Id;
    }

    public async Task SubmitForReviewAsync(long quotationId, long techReviewerId)
    {
        var quote = await _db.Quotations.FindAsync(quotationId);
        if (quote == null || quote.Status != 0)
            throw new BusinessException("报价单状态不正确，无法提交评审");

        quote.Status = 1; // 待评审
        quote.TechReviewBy = techReviewerId;
        quote.UpdatedAt = DateTime.Now;
        await _db.SaveChangesAsync();
    }

    public async Task TechReviewAsync(long quotationId, TechReviewDto dto)
    {
        var quote = await _db.Quotations.FindAsync(quotationId);
        if (quote == null || quote.Status != 1)
            throw new BusinessException("报价单状态不正确");

        quote.TechReviewBy = dto.ReviewerId;
        quote.TechReviewDate = DateTime.Now;
        quote.TechReviewCost = dto.Cost;
        quote.TechReviewNote = dto.Note;
        quote.Status = dto.Approved ? 2 : 0; // 通过=待审批，驳回=草稿
        quote.UpdatedAt = DateTime.Now;

        await _db.SaveChangesAsync();

        _logger.LogInformation("报价单技术评审: Id={Id}, 结果={Approved}", quotationId, dto.Approved);
    }

    public async Task PricingApproveAsync(long quotationId, PricingApproveDto dto)
    {
        var quote = await _db.Quotations.FindAsync(quotationId);
        if (quote == null || quote.Status != 2)
            throw new BusinessException("报价单状态不正确");

        quote.PricingApprovedBy = dto.ApproverId;
        quote.PricingApprovedDate = DateTime.Now;
        quote.PricingNote = dto.Note;
        quote.Status = dto.Approved ? 3 : 1; // 批准=已发出，驳回=待评审
        quote.UpdatedAt = DateTime.Now;

        await _db.SaveChangesAsync();

        _logger.LogInformation("报价单定价审批: Id={Id}, 结果={Approved}", quotationId, dto.Approved);
    }

    public async Task<QuoQuotation?> GetQuotationDetailAsync(long id)
    {
        return await _db.Quotations
            .AsNoTracking()
            .Include(q => q.Items)
            .Include(q => q.Opportunity)
            .FirstOrDefaultAsync(q => q.Id == id);
    }

    public async Task<PagedResult<QuoOpportunity>> GetOpportunityListAsync(
        int pageNum, int pageSize, string? keyword)
    {
        var query = _db.Opportunities.AsNoTracking().AsQueryable();

        if (!string.IsNullOrWhiteSpace(keyword))
        {
            query = query.Where(o => o.Title.Contains(keyword) || o.OpportunityNo.Contains(keyword));
        }

        var total = await query.CountAsync();
        var list = await query
            .OrderByDescending(o => o.CreatedAt)
            .Skip((pageNum - 1) * pageSize)
            .Take(pageSize)
            .ToListAsync();

        return PagedResult<QuoOpportunity>.Of(list, total, pageNum, pageSize);
    }
}
```

```csharp
using CpmServer.Common;
using CpmServer.DTOs.Quotation;
using CpmServer.Models;

using Microsoft.AspNetCore.Authorization;

namespace CpmServer.Controllers;

[ApiController]
[Route("api/[controller]")]

        _quotationService = quotationService;

        return ApiResult<long>.Success(id);

    }

    public async Task<ApiResult> SubmitForReview(long id, [FromQuery] long techReviewerId)
    {
        await _quotationService.SubmitForReviewAsync(id, techReviewerId);

    }

    public async Task<ApiResult> TechReview(long id, [FromBody] TechReviewDto dto)

        await _quotationService.TechReviewAsync(id, dto);

    }

    [HttpPost("{id}/pricing-approve")]
    public async Task<ApiResult> PricingApprove(long id, [FromBody] PricingApproveDto dto)
    {
        await _quotationService.PricingApproveAsync(id, dto);
        return ApiResult.Success();
```

```vue
      <div class="status-row">
        <span class="label">当前状态：</span>
        <el-tag :type="statusType" size="large">{{ statusLabel }}</el-tag>
        <span class="label" style="margin-left: 40px">总金额：</span>

      </div>
    </el-card>

    <!-- 报价明细 -->
    <el-card>
      <template #header>报价明细</template>
      <el-table :data="detail?.items || []" border>
        <el-table-column type="index" label="序号" />
        <el-table-column prop="productId" label="产品ID" />
        <el-table-column prop="qty" label="数量" />
        <el-table-column prop="unitPrice" label="单价" />
        <el-table-column prop="lineAmount" label="金额" />
      </el-table>
    </el-card>

    <!-- 技术评审区 -->
    <el-card v-if="detail?.status >= 1">
      <template #header>技术评审</template>
      <div v-if="detail?.status === 1">
        <el-alert title="待技术评审" type="warning" :closable="false" />
        <el-form v-if="isTechUser" :model="reviewForm" label-width="100px" style="margin-top: 16px">
          <el-form-item label="评估成本">
            <el-input-number v-model="reviewForm.cost" :min="0" :precision="2" />

      </div>
      <div v-else>

        <p><strong>评审意见：</strong>{{ detail?.techReviewNote }}</p>
      </div>

    <!-- 定价审批区 -->
    <el-card v-if="detail?.status >= 2">
      <template #header>定价审批</template>
      <div v-if="detail?.status === 2">

        <el-form v-if="isManager" :model="approveForm" label-width="100px" style="margin-top: 16px">

            <el-button type="danger" @click="handleApprove(false)">驳回</el-button>
          </el-form-item>
        </el-form>
        <p v-else>当前报价单正在等待定价审批</p>
      </div>
      <div v-else>
        <p><strong>审批人ID：</strong>{{ detail?.pricingApprovedBy }}</p>
        <p><strong>审批时间：</strong>{{ detail?.pricingApprovedDate }}</p>
        <p><strong>审批意见：</strong>{{ detail?.pricingNote }}</p>
      </div>
    </el-card>
  </div>

<script setup lang="ts">
import { ref, computed, onMounted } from 'vue'
import { useRoute } from 'vue-router'
import { quotationApi } from '@/api/quotation'
import { useUserStore } from '@/stores/user'

const route = useRoute()
const userStore = useUserStore()
const loading = ref(false)
const detail = ref<any>({})
const reviewForm = ref({ cost: 0, note: '' })
const approveForm = ref({ note: '' })

const id = Number(route.params.id)

const isTechUser = computed(() => userStore.roles.includes('TECH'))
const isManager = computed(() => userStore.roles.includes('MANAGER'))

const statusType = computed(() => {
  const map: Record<number, string> = { 0: 'info', 1: 'warning', 2: 'warning', 3: 'success', 9: 'success' }
  return map[detail.value.status] || 'info'
})
const statusLabel = computed(() => {
  const map: Record<number, string> = { 0: '草稿', 1: '待技术评审', 2: '待定价审批', 3: '已发出', 9: '完成' }
  return map[detail.value.status] || '未知'
})

const loadDetail = async () => {
  loading.value = true
  const res = await quotationApi.getQuotation(id)
  detail.value = res.data
  loading.value = false
}

const handleReview = async (approved: boolean) => {
  await quotationApi.techReview(id, {
    reviewerId: userStore.userInfo.id,
    approved,
    cost: reviewForm.value.cost,
    note: reviewForm.value.note
  })
  loadDetail()
}

const handleApprove = async (approved: boolean) => {
  await quotationApi.pricingApprove(id, {
    approverId: userStore.userInfo.id,
    approved,
    note: approveForm.value.note
  })
  loadDetail()
}

onMounted(loadDetail)
</script>
```
