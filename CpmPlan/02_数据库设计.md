# 数据库设计（核心表）

> 按阶段逐步添加表，不要一次全建。当前列出的是P1-P9所需的核心表。
> 数据库：SQL Server 2022

---

## 命名规范

- 表名：`模块_表名`，如 `SysUser`、`QuoOpportunity`
- C#实体：Pascal命名，如 `SysUser`、`SampleBatch`
- 主键：统一叫 `Id`，bigint，自增
- 时间字段：`CreatedAt`、`UpdatedAt`
- 逻辑删除：`IsDeleted`，bit，0=未删除，1=已删除
- 状态字段：`Status`，int：0=草稿，1=待处理，2=处理中，9=完成/关闭
- 金额：decimal(18,2)
- 时间类型：`datetime2`

    RealName NVARCHAR(50),
    Phone NVARCHAR(20),
    Email NVARCHAR(100),
    DeptId BIGINT,
    IsActive BIT DEFAULT 1,
    CreatedAt DATETIME2 DEFAULT GETDATE(),

);

CREATE TABLE SysRole (
    Id BIGINT IDENTITY(1,1) PRIMARY KEY,

public class CpmDbContext : DbContext
{
    public CpmDbContext(DbContextOptions<CpmDbContext> options) : base(options) { }
    
    public DbSet<SysUser> Users => Set<SysUser>();
    public DbSet<SysRole> Roles => Set<SysRole>();
    public DbSet<SysUserRole> UserRoles => Set<SysUserRole>();
    public DbSet<SysDept> Depts => Set<SysDept>();
    
    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        modelBuilder.Entity<SysUser>(entity =>
        {
            entity.ToTable("SysUser");
            entity.HasIndex(e => e.Username).IsUnique();
        });
        
        modelBuilder.Entity<SysUserRole>(entity =>
        {
            entity.HasKey(e => new { e.UserId, e.RoleId });
        });
    }
}
